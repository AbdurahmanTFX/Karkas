file(GLOB KK_CLI_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
add_executable(karkas ${KK_CLI_SOURCES})
target_link_libraries(karkas PRIVATE CLI11::CLI11 spdlog::spdlog)
foreach(lib core index code runner llm metrics agents mcp layout)
  if(TARGET kk_${lib})
    target_link_libraries(karkas PRIVATE kk_${lib})
    string(TOUPPER ${lib} _LIB)
    target_compile_definitions(karkas PRIVATE KK_HAVE_${_LIB}=1)
  endif()
endforeach()
target_compile_definitions(karkas PRIVATE KARKAS_VERSION="${PROJECT_VERSION}" KK_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
kk_set_warnings(karkas)
